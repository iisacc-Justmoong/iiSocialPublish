cmake_minimum_required(VERSION 3.24)
project(iiSocialPublish VERSION 0.1.1 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
include(CTest)
find_package(CURL 7.68 REQUIRED)
find_package(Threads REQUIRED)
add_library(iiSocialPublish STATIC src/Publishing.cpp src/CurlTransport.cpp)
add_library(iiSocialPublish::iiSocialPublish ALIAS iiSocialPublish)
set_target_properties(iiSocialPublish PROPERTIES CXX_EXTENSIONS OFF POSITION_INDEPENDENT_CODE ON)
target_compile_features(iiSocialPublish PUBLIC cxx_std_23)
target_include_directories(iiSocialPublish PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>
    PRIVATE third_party)
target_link_libraries(iiSocialPublish PRIVATE CURL::libcurl Threads::Threads)
if(MSVC)
    target_compile_options(iiSocialPublish PRIVATE /W4 /permissive-)
else()
    target_compile_options(iiSocialPublish PRIVATE -Wall -Wextra -Wpedantic)
endif()
set(_package_dir "${CMAKE_INSTALL_LIBDIR}/cmake/iiSocialPublish")
install(TARGETS iiSocialPublish EXPORT iiSocialPublishTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT iiSocialPublishTargets NAMESPACE iiSocialPublish:: DESTINATION "${_package_dir}")
configure_package_config_file(cmake/iiSocialPublishConfig.cmake.in
    "${CMAKE_CURRENT_BINARY_DIR}/iiSocialPublishConfig.cmake" INSTALL_DESTINATION "${_package_dir}")
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/iiSocialPublishConfigVersion.cmake"
    VERSION ${PROJECT_VERSION} COMPATIBILITY SameMinorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/iiSocialPublishConfig.cmake"
    "${CMAKE_CURRENT_BINARY_DIR}/iiSocialPublishConfigVersion.cmake" DESTINATION "${_package_dir}")
install(FILES README.md DESTINATION ${CMAKE_INSTALL_DATADIR}/iiSocialPublish)
install(DIRECTORY docs/ DESTINATION ${CMAKE_INSTALL_DATADIR}/iiSocialPublish/docs)
install(DIRECTORY third_party/licenses/ DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/iiSocialPublish)
install(FILES third_party/dependencies.json DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/iiSocialPublish)
if(BUILD_TESTING)
    add_executable(iiSocialPublishTests tests/PublishingTest.cpp)
    target_link_libraries(iiSocialPublishTests PRIVATE iiSocialPublish::iiSocialPublish)
    target_include_directories(iiSocialPublishTests PRIVATE third_party)
    foreach(_case catalog validation x facebook instagram threads pinterest tumblr mastodon linkedin errors transport)
        add_test(NAME iiSocialPublish.${_case} COMMAND iiSocialPublishTests ${_case})
    endforeach()
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    find_program(IISOCIALPUBLISH_OPENSSL openssl REQUIRED)
    add_test(NAME iiSocialPublish.TlsLoopback
        COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/TransportLoopback.py"
            "$<TARGET_FILE:iiSocialPublishTests>" "${CMAKE_CURRENT_BINARY_DIR}/tls-loopback" "${IISOCIALPUBLISH_OPENSSL}")
    set_tests_properties(iiSocialPublish.TlsLoopback PROPERTIES TIMEOUT 60)
endif()
